  v/.style={circle,draw,inner sep=1.2pt,font=\small,fill=white}]

  \node[lab] at (-3.2,-0.2) {the graph $G$};
  % left branch (nodes 0..5)
  \foreach \i/\b/\y in {0/{$\emptyset$}/0, 1/{$\{0\}$}/1, 2/{$\{0,2\}$}/2, 3/{$\{0,1,2\}$}/3, 4/{$\{1,2\}$}/4, 5/{$\{2\}$}/5}
